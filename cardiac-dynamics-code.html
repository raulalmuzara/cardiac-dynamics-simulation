d = 2 # Number of equations
ti = 0 # Initial time
tf = 5 # Final time
h = 0.0001 # Step size
epsilon = 0.04 # Relates fast and slow time scales in fast-slow systems

t = np.zeros((int((tf-ti)/h)+1 , 1)) # Time variable
y = np.zeros((int((tf-ti)/h)+1 , d)) # y[0] is "x" // y[1] is "y"

k1 = np.zeros((int((tf-ti)/h)+1 , d)) # Slope 1
k2 = np.zeros((int((tf-ti)/h)+1 , d)) # Slope 2
k3 = np.zeros((int((tf-ti)/h)+1 , d)) # Slope 3
k4 = np.zeros((int((tf-ti)/h)+1 , d)) # Slope 4

a = 0.5 #0.9948377923 #0.9934909321 # Control parameter

def equations(t,y):
    return [ (y[1]-y[0]**3/3+y[0])/epsilon , a-y[0] ]

#Initial conditions:

t0 = ti
y0 = [2,0]

t[0] = t0
y[0] = y0

# Runge-Kutta fourth-order method:

for i in range(1,int((tf-ti)/h)+1):
    
    k1[i-1] = equations(t[i-1] , y[i-1])
    k2[i-1] = equations(t[i-1] + h/2 , y[i-1] + h*k1[i-1]/2)
    k3[i-1] = equations(t[i-1] + h/2 , y[i-1] + h*k2[i-1]/2)
    k4[i-1] = equations(t[i-1] + h , y[i-1] + h*k3[i-1])
    
    y[i] = y[i-1] + h*(k1[i-1] + 2*k2[i-1] + 2*k3[i-1] + k4[i-1])/6
    t[i] = t[i-1] + h

ti = 0 # Initial time
tf = 600 # Final time
h = 0.01 # Time step

y = np.zeros((int((tf-ti)/h)+1 , d))
t = np.zeros((int((tf-ti)/h)+1 , 1))

k1 = np.zeros((int((tf-ti)/h)+1 , d)) # Slope 1
k2 = np.zeros((int((tf-ti)/h)+1 , d)) # Slope 2
k3 = np.zeros((int((tf-ti)/h)+1 , d)) # Slope 3
k4 = np.zeros((int((tf-ti)/h)+1 , d)) # Slope 4


def equations(t,y):
    
    # t is the independent fast time scale variable
    # y[0] is E, the transmembrane potential
    # y[1] is n, the slow gating variable for ion channels
    
    Estar = 1.5
    M = 4
    epsilon = 0.01
    nB = 0.5
    I = 0

    #def stepfunction(E):
    #    if E <= 0:
    #        return 0
    #    else:
    #        return 1
    
    def stepfunction(E):
        return max(0,E)
        
    equation1 = -y[0] + 2 * (Estar - y[1]**M) * (y[0]**2 - (y[0]**3)/4) + I
    equation2 = epsilon * (stepfunction(y[0]-1)/nB - y[1])
    
    return [equation1,equation2]

# Initial conditions:

t0 = ti
y0 = [0.5,0.5]

t[0] = t0
y[0] = y0

# Runge-Kutta fourth-order method:

for i in range(1,int((tf-ti)/h)+1):
    
    k1[i-1] = equations(t[i-1] , y[i-1])
    k2[i-1] = equations(t[i-1] + h/2 , y[i-1] + h*k1[i-1]/2)
    k3[i-1] = equations(t[i-1] + h/2 , y[i-1] + h*k2[i-1]/2)
    k4[i-1] = equations(t[i-1] + h , y[i-1] + h*k3[i-1])
    
    y[i] = y[i-1] + h*(k1[i-1] + 2*k2[i-1] + 2*k3[i-1] + k4[i-1])/6
    t[i] = t[i-1] + h

Simulation of the reduced Luo-Rudy model¶

d = 3 # Number of differential equations
ti = 0 # Initial time
tf = 8000 # Final time
h = 0.01 # Time step

y = np.zeros((int((tf-ti)/h)+1 , d))
t = np.zeros((int((tf-ti)/h)+1 , 1))

k1 = np.zeros((int((tf-ti)/h)+1 , d)) # Slope 1
k2 = np.zeros((int((tf-ti)/h)+1 , d)) # Slope 2
k3 = np.zeros((int((tf-ti)/h)+1 , d)) # Slope 3
k4 = np.zeros((int((tf-ti)/h)+1 , d)) # Slope 4


def equations(t,y):
    
    # t is the independent fast time scale variable
    # y[0] is V, the transmembrane potential
    # y[1] is f, the inactivation gating variable associated to the Ca current
    # y[2] is x, the activation gating variable associated to the K current
    
    Cm = 0.5
    tauf = 80
    taux = 300
    GCa = 0.025
    ECa = 100
    GK = 0.05
    EK = -80
    Ef = -20
    thetaf = 8.6
    Ex = -40
    thetax = 5
    Ed = -35
    thetad = 6.24
    Isti = 0
    
    def finfinity(V):
        return 1 / (1 + np.exp((V-Ef)/(thetaf)))

    def xinfinity(V):
        return 1 / (1 + np.exp((Ex-V)/(thetax)))

    def dinfinity(V):
        return 1 / (1 + np.exp((Ed-V)/(thetad)))
    
    def ICa(V,f):
        return GCa * dinfinity(V) * f * (V-ECa)
    
    def IK(V,x):
        return GK * x * (V-EK)
    
    equation1 = -(ICa(y[0],y[1])+IK(y[0],y[2])+Isti)/(Cm)
    equation2 = (finfinity(y[0])-y[1])/(tauf)
    equation3 = (xinfinity(y[0])-y[2])/(taux)
    
    return [equation1,equation2,equation3]

# Initial conditions:

t0 = ti
y0 = [0,0.5,0.5]

t[0] = t0
y[0] = y0

# Runge-Kutta fourth-order method:

for i in range(1,int((tf-ti)/h)+1):
    
    k1[i-1] = equations(t[i-1] , y[i-1])
    k2[i-1] = equations(t[i-1] + h/2 , y[i-1] + h*k1[i-1]/2)
    k3[i-1] = equations(t[i-1] + h/2 , y[i-1] + h*k2[i-1]/2)
    k4[i-1] = equations(t[i-1] + h , y[i-1] + h*k3[i-1])
    
    y[i] = y[i-1] + h*(k1[i-1] + 2*k2[i-1] + 2*k3[i-1] + k4[i-1])/6
    t[i] = t[i-1] + h
